---
title: useIsomorphicEffect
description: A cross-platform effect hook aliasing `useLayoutEffect` on the client.
---

```tsx
import { useIsomorphicEffect } from "@piplup/utils";

function Component() {
  useIsomorphicEffect(() => {
    // runs like useLayoutEffect in client environments
  }, []);

  return null;
}
```
